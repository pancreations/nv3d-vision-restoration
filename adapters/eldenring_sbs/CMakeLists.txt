cmake_minimum_required(VERSION 3.24)
project(VisionEldenSBS LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_compile_definitions(UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
add_subdirectory("${ROOT}/third_party/minhook-1.3.4" minhook)
foreach(target VisionEldenSBS VisionEldenSBSTest)
  add_library(${target} MODULE addon.cpp)
  target_include_directories(${target} PRIVATE "${ROOT}/third_party/reshade-main/include")
  target_link_libraries(${target} PRIVATE minhook d3dcompiler dxgi version)
  target_compile_options(${target} PRIVATE /W4 /EHsc /utf-8 /permissive-)
  set_target_properties(${target} PROPERTIES SUFFIX ".addon64")
endforeach()
target_compile_definitions(VisionEldenSBSTest PRIVATE VISION_ELDEN_TEST)
add_executable(vision_elden_tests tests.cpp)
target_link_libraries(vision_elden_tests PRIVATE d3d11 d3dcompiler dxgi)
target_compile_options(vision_elden_tests PRIVATE /W4 /EHsc /utf-8 /permissive-)
enable_testing()
add_test(NAME elden_camera_and_sbs COMMAND vision_elden_tests)
add_executable(vision_elden_dx12_test dx12_test.cpp)
target_link_libraries(vision_elden_dx12_test PRIVATE d3d12 d3dcompiler dxgi)
target_compile_options(vision_elden_dx12_test PRIVATE /W4 /EHsc /utf-8 /permissive-)
