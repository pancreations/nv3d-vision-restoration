cmake_minimum_required(VERSION 3.24)
project(VisionStereoRuntime LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
add_compile_definitions(UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
add_library(VisionStereoLoader SHARED ../../runtime/geo11_loader.cpp ../../runtime/geo11_loader.def)
target_link_libraries(VisionStereoLoader PRIVATE dxgi)
target_compile_options(VisionStereoLoader PRIVATE /W4 /EHsc /utf-8)
add_executable(vision_geo11_probe geo11_probe.cpp)
target_link_libraries(vision_geo11_probe PRIVATE d3d11 d3dcompiler dxgi user32)
target_compile_options(vision_geo11_probe PRIVATE /W4 /EHsc /utf-8)
add_subdirectory(../../third_party/minhook-1.3.4 "${CMAKE_CURRENT_BINARY_DIR}/minhook")
add_library(VisionStereo11 SHARED ../../runtime/geo11_output.cpp ../../runtime/geo11_output.def)
target_include_directories(VisionStereo11 PRIVATE ../../src ../../third_party/minhook-1.3.4/include)
target_link_libraries(VisionStereo11 PRIVATE minhook d3dcompiler dxgi user32 avrt)
target_compile_options(VisionStereo11 PRIVATE /W4 /EHsc /utf-8)
add_executable(vision_debug_run debug_run.cpp)
target_link_libraries(vision_debug_run PRIVATE dbghelp)
target_compile_options(vision_debug_run PRIVATE /W4 /EHsc /utf-8)
add_library(VisionStereo11Test SHARED ../../runtime/geo11_output.cpp ../../runtime/geo11_output.def)
target_compile_definitions(VisionStereo11Test PRIVATE VISION_STEREO_TESTING=1)
target_include_directories(VisionStereo11Test PRIVATE ../../src ../../third_party/minhook-1.3.4/include)
target_link_libraries(VisionStereo11Test PRIVATE minhook d3dcompiler dxgi user32 avrt)
target_compile_options(VisionStereo11Test PRIVATE /W4 /EHsc /utf-8)
add_executable(vision_output_tests output_tests.cpp)
target_include_directories(vision_output_tests PRIVATE ../../src ../../runtime)
target_link_libraries(vision_output_tests PRIVATE d3d11 dxgi user32)
target_compile_options(vision_output_tests PRIVATE /W4 /EHsc /utf-8)
add_executable(vision_pair_inspect pair_inspect.cpp)
target_link_libraries(vision_pair_inspect PRIVATE d3d11 dxgi)
target_compile_options(vision_pair_inspect PRIVATE /W4 /EHsc /utf-8)
add_executable(vision_sync_inspect sync_inspect.cpp)
target_include_directories(vision_sync_inspect PRIVATE ../../src)
target_compile_options(vision_sync_inspect PRIVATE /W4 /EHsc /utf-8)
