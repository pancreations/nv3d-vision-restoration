cmake_minimum_required(VERSION 3.24)
project(VisionRestoration VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "This application targets Windows 10/11.")
endif()
add_compile_definitions(UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(IMGUI "${CMAKE_SOURCE_DIR}/third_party/imgui-1.91.9b")
add_library(imgui STATIC ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp ${IMGUI}/imgui_tables.cpp ${IMGUI}/imgui_widgets.cpp ${IMGUI}/backends/imgui_impl_win32.cpp ${IMGUI}/backends/imgui_impl_dx11.cpp)
target_include_directories(imgui PUBLIC ${IMGUI} ${IMGUI}/backends)
target_link_libraries(imgui PUBLIC d3d11 dxgi d3dcompiler dwmapi)
add_library(vision_core STATIC src/core.cpp src/lcd_timing.cpp src/firmware.cpp src/panel_experiments.cpp src/screen_depth.cpp)
target_include_directories(vision_core PUBLIC src)
add_library(vision_compatibility STATIC src/stereo_compatibility.cpp)
target_include_directories(vision_compatibility PUBLIC src)
target_compile_options(vision_compatibility PRIVATE /W4 /EHsc /utf-8 /permissive-)
add_executable(vision_compile_shaders tools/compile_shaders.cpp)
target_link_libraries(vision_compile_shaders PRIVATE d3dcompiler)
set(STEREO_SHADER_HEADER "${CMAKE_BINARY_DIR}/generated/stereo_shader_bytecode.h")
add_custom_command(OUTPUT "${STEREO_SHADER_HEADER}"
  COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/generated"
  COMMAND $<TARGET_FILE:vision_compile_shaders> "${CMAKE_SOURCE_DIR}/src/stereo_scene.hlsl" "${STEREO_SHADER_HEADER}"
  DEPENDS vision_compile_shaders "${CMAKE_SOURCE_DIR}/src/stereo_scene.hlsl"
  COMMENT "Compiling stereo shaders at build time" VERBATIM)
add_library(vision_platform STATIC src/display.cpp src/emitter.cpp src/emitter_rp2040.cpp src/rp2040_usb.cpp src/renderer.cpp src/sources.cpp src/screen_source.cpp src/gamesync.cpp)
target_sources(vision_platform PRIVATE "${STEREO_SHADER_HEADER}")
target_include_directories(vision_platform PRIVATE "${CMAKE_BINARY_DIR}/generated")
target_include_directories(vision_platform PUBLIC src third_party/libusb/include)
target_link_libraries(vision_platform PUBLIC vision_core imgui d3d11 dxgi d3dcompiler windowsapp windowscodecs setupapi cfgmgr32 avrt shcore comdlg32 "${CMAKE_SOURCE_DIR}/third_party/libusb/VS2022/MS64/dll/libusb-1.0.lib")
add_executable(VisionRestoration WIN32 src/main.cpp src/lcd_ui.cpp src/key_bindings.cpp)
target_link_libraries(VisionRestoration PRIVATE vision_platform vision_compatibility dbghelp)
add_executable(vision_usb_recover WIN32 tools/usb_port_recover.cpp)
target_link_libraries(vision_usb_recover PRIVATE setupapi cfgmgr32)
add_dependencies(VisionRestoration vision_usb_recover)
add_custom_command(TARGET VisionRestoration POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/third_party/libusb/VS2022/MS64/dll/libusb-1.0.dll" "$<TARGET_FILE_DIR:VisionRestoration>")
# Optional, user-supplied: a locally built libwdi installer and the proprietary emitter firmware are not in the repository.
if(EXISTS "${CMAKE_SOURCE_DIR}/third_party/libwdi/bin/wdi-simple-x64.exe")
  add_custom_command(TARGET VisionRestoration POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/third_party/libwdi/bin/wdi-simple-x64.exe" "$<TARGET_FILE_DIR:VisionRestoration>/vision_winusb_installer.exe")
endif()
if(EXISTS "${CMAKE_SOURCE_DIR}/STUFF/emitter.fw")
  add_custom_command(TARGET VisionRestoration POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/STUFF/emitter.fw" "$<TARGET_FILE_DIR:VisionRestoration>/emitter.fw")
endif()
add_executable(vision_tests tests/core_tests.cpp)
target_link_libraries(vision_tests PRIVATE vision_core)
enable_testing()
add_test(NAME core COMMAND vision_tests)

target_link_libraries(vision_compatibility_tests PRIVATE vision_compatibility)
add_test(NAME existing_stereo_fix_connection COMMAND vision_compatibility_tests)
add_executable(vision_stereo_setup tools/stereo_setup.cpp)
target_link_libraries(vision_stereo_setup PRIVATE vision_compatibility)
add_executable(vision_lcd_tests tests/lcd_timing_tests.cpp)
target_link_libraries(vision_lcd_tests PRIVATE vision_core)
add_test(NAME lcd_temporal_aperture COMMAND vision_lcd_tests)
add_executable(vision_controls_tests tests/controls_tests.cpp src/lcd_ui.cpp)
target_link_libraries(vision_controls_tests PRIVATE vision_core imgui)
add_test(NAME calibration_controls COMMAND vision_controls_tests)
add_executable(vision_menu_tests tests/menu_tests.cpp src/key_bindings.cpp)
target_include_directories(vision_menu_tests PRIVATE src)
target_link_libraries(vision_menu_tests PRIVATE user32)
add_test(NAME fullscreen_menu_shortcuts COMMAND vision_menu_tests)
add_executable(vision_depth_tests tests/depth_tests.cpp)
target_link_libraries(vision_depth_tests PRIVATE vision_core)

add_executable(vision_screen_tests tests/screen_tests.cpp)

add_custom_command(TARGET vision_screen_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/third_party/libusb/VS2022/MS64/dll/libusb-1.0.dll" "$<TARGET_FILE_DIR:vision_screen_tests>")
add_test(NAME screen_conversion_shader COMMAND vision_screen_tests)
set_tests_properties(screen_conversion_shader PROPERTIES TIMEOUT 60)
# Presentation on a capture-excluded, layered overlay window (a tool, not a test).
add_executable(vision_overlay_probe tools/overlay_probe.cpp)
target_link_libraries(vision_overlay_probe PRIVATE vision_platform)
add_custom_command(TARGET vision_overlay_probe POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/third_party/libusb/VS2022/MS64/dll/libusb-1.0.dll" "$<TARGET_FILE_DIR:vision_overlay_probe>")
# GPU cost of the conversion at 4K on the installed GPU (a tool, not a test).
add_executable(vision_screen_bench tools/screen_bench.cpp)
target_link_libraries(vision_screen_bench PRIVATE vision_platform)
add_custom_command(TARGET vision_screen_bench POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/third_party/libusb/VS2022/MS64/dll/libusb-1.0.dll" "$<TARGET_FILE_DIR:vision_screen_bench>")
# Depth helper for the whole-screen conversion. ONNX Runtime (DirectML build) and DirectML come
# from tools/Get-Dependencies.ps1; without them the app builds and the screen input reports
# that the helper is missing. onnxruntime.dll and DirectML.dll are loaded at run time.
set(ORT_DIR "${CMAKE_SOURCE_DIR}/third_party/onnxruntime")
set(DML_DIR "${CMAKE_SOURCE_DIR}/third_party/directml")
if(EXISTS "${ORT_DIR}/build/native/include/onnxruntime_c_api.h" AND EXISTS "${DML_DIR}/include/DirectML.h")
  add_executable(VisionDepth tools/depth_helper.cpp)
  target_include_directories(VisionDepth PRIVATE src "${ORT_DIR}/build/native/include" "${DML_DIR}/include")
  target_link_libraries(VisionDepth PRIVATE vision_core d3d12 dxgi)
  target_compile_options(VisionDepth PRIVATE /W4 /EHsc /utf-8 /permissive-)
  add_custom_command(TARGET VisionDepth POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ORT_DIR}/runtimes/win-x64/native/onnxruntime.dll" "${DML_DIR}/bin/x64-win/DirectML.dll" "$<TARGET_FILE_DIR:VisionDepth>")
  add_dependencies(VisionRestoration VisionDepth)
else()
  message(STATUS "ONNX Runtime / DirectML not found under third_party: VisionDepth (AI screen depth) is not built. Run tools/Get-Dependencies.ps1.")
endif()
