# Depth helper for the whole-screen conversion. ONNX Runtime (DirectML build) and DirectML come
# from tools/Get-Dependencies.ps1; without them the app builds and the screen input reports
# that the helper is missing. onnxruntime.dll and DirectML.dll are loaded at run time.
set(ORT_DIR "${CMAKE_SOURCE_DIR}/third_party/onnxruntime")
set(DML_DIR "${CMAKE_SOURCE_DIR}/third_party/directml")
if(EXISTS "${ORT_DIR}/build/native/include/onnxruntime_c_api.h" AND EXISTS "${DML_DIR}/include/DirectML.h")
  add_executable(VisionDepth tools/depth_helper.cpp)
  target_include_directories(VisionDepth PRIVATE src "${ORT_DIR}/build/native/include" "${DML_DIR}/include")
  target_link_libraries(VisionDepth PRIVATE vision_core d3d12 dxgi)
  target_compile_options(VisionDepth PRIVATE /W4 /EHsc /utf-8 /permissive-)
  add_custom_command(TARGET VisionDepth POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ORT_DIR}/runtimes/win-x64/native/onnxruntime.dll" "${DML_DIR}/bin/x64-win/DirectML.dll" "$<TARGET_FILE_DIR:VisionDepth>")
  add_dependencies(VisionRestoration VisionDepth)
else()
  message(STATUS "ONNX Runtime / DirectML not found under third_party: VisionDepth (AI screen depth) is not built. Run tools/Get-Dependencies.ps1.")
endif()
