cmake_minimum_required(VERSION 3.24)
project(VisionRestoration VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "This application targets Windows 10/11.")
endif()
add_compile_definitions(UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(IMGUI "${CMAKE_SOURCE_DIR}/third_party/imgui-1.91.9b")
add_library(imgui STATIC ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp ${IMGUI}/imgui_tables.cpp ${IMGUI}/imgui_widgets.cpp ${IMGUI}/backends/imgui_impl_win32.cpp ${IMGUI}/backends/imgui_impl_dx11.cpp)
target_include_directories(imgui PUBLIC ${IMGUI} ${IMGUI}/backends)
target_link_libraries(imgui PUBLIC d3d11 dxgi d3dcompiler dwmapi)
set(SPOUT_BUILD_CMT OFF CACHE BOOL "" FORCE)
set(SPOUT_BUILD_LIBRARY OFF CACHE BOOL "" FORCE)
set(SPOUT_BUILD_SPOUTDX ON CACHE BOOL "" FORCE)
set(SKIP_INSTALL_ALL ON CACHE BOOL "" FORCE)
add_subdirectory(third_party/Spout2-master EXCLUDE_FROM_ALL)
add_library(vision_core STATIC src/core.cpp src/firmware.cpp)
target_include_directories(vision_core PUBLIC src)
add_library(vision_platform STATIC src/display.cpp src/emitter.cpp src/emitter_rp2040.cpp src/rp2040_usb.cpp src/renderer.cpp src/sources.cpp src/gamesync.cpp)
target_include_directories(vision_platform PUBLIC src third_party/libusb/include third_party/Spout2-master/SPOUTSDK/SpoutDirectX/SpoutDX)
target_compile_definitions(vision_platform PUBLIC SPOUT_BUILD_STATIC)
target_link_libraries(vision_platform PUBLIC vision_core imgui SpoutDX_static d3d11 dxgi d3dcompiler windowsapp windowscodecs setupapi cfgmgr32 avrt shcore comdlg32 "${CMAKE_SOURCE_DIR}/third_party/libusb/VS2022/MS64/dll/libusb-1.0.lib")
add_executable(VisionRestoration WIN32 src/main.cpp)
target_link_libraries(VisionRestoration PRIVATE vision_platform)
add_executable(vision_usb_recover WIN32 tools/usb_port_recover.cpp)
target_link_libraries(vision_usb_recover PRIVATE setupapi cfgmgr32)
add_dependencies(VisionRestoration vision_usb_recover)
add_custom_command(TARGET VisionRestoration POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/third_party/libusb/VS2022/MS64/dll/libusb-1.0.dll" "$<TARGET_FILE_DIR:VisionRestoration>")
# Optional, user-supplied: a locally built libwdi installer and the proprietary emitter firmware are not in the repository.
if(EXISTS "${CMAKE_SOURCE_DIR}/third_party/libwdi/bin/wdi-simple-x64.exe")
  add_custom_command(TARGET VisionRestoration POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/third_party/libwdi/bin/wdi-simple-x64.exe" "$<TARGET_FILE_DIR:VisionRestoration>/vision_winusb_installer.exe")
endif()
if(EXISTS "${CMAKE_SOURCE_DIR}/STUFF/emitter.fw")
  add_custom_command(TARGET VisionRestoration POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/STUFF/emitter.fw" "$<TARGET_FILE_DIR:VisionRestoration>/emitter.fw")
endif()
add_executable(vision_tests tests/core_tests.cpp)
target_link_libraries(vision_tests PRIVATE vision_core)
enable_testing()
add_test(NAME core COMMAND vision_tests)
add_library(vision_rp2040 STATIC src/rp2040_protocol.cpp src/rp2040_wave.cpp src/rp2040_host.cpp)
target_link_libraries(vision_platform PUBLIC vision_rp2040)
target_include_directories(vision_rp2040 PUBLIC src)
add_executable(vision_rp2040_tests tests/rp2040_tests.cpp)
target_link_libraries(vision_rp2040_tests PRIVATE vision_rp2040)
add_test(NAME rp2040_protocol_scheduler COMMAND vision_rp2040_tests)
set_tests_properties(rp2040_protocol_scheduler PROPERTIES TIMEOUT 30)
add_executable(vision_rp2040_host_tests tests/rp2040_host_tests.cpp)
target_link_libraries(vision_rp2040_host_tests PRIVATE vision_rp2040)
add_test(NAME rp2040_host_and_waveforms COMMAND vision_rp2040_host_tests)
add_executable(vision_rp2040_probe tools/rp2040_probe.cpp)
target_link_libraries(vision_rp2040_probe PRIVATE vision_platform)
add_custom_command(TARGET vision_rp2040_probe POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/third_party/libusb/VS2022/MS64/dll/libusb-1.0.dll" "$<TARGET_FILE_DIR:vision_rp2040_probe>")
add_executable(vision_platform_tests tests/platform_tests.cpp)
target_link_libraries(vision_platform_tests PRIVATE vision_platform)
add_custom_command(TARGET vision_platform_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/third_party/libusb/VS2022/MS64/dll/libusb-1.0.dll" "$<TARGET_FILE_DIR:vision_platform_tests>")
add_test(NAME gpu_texture_integration COMMAND vision_platform_tests)
set_tests_properties(gpu_texture_integration PROPERTIES TIMEOUT 30)
add_executable(vision_firmware tools/firmware_main.cpp)
target_link_libraries(vision_firmware PRIVATE vision_core)
add_library(VisionStereoSpout MODULE adapters/reshade_spout.cpp)
target_include_directories(VisionStereoSpout PRIVATE third_party/reshade-main/include third_party/Spout2-master/SPOUTSDK/SpoutDirectX/SpoutDX)
target_compile_definitions(VisionStereoSpout PRIVATE SPOUT_BUILD_STATIC)
target_link_libraries(VisionStereoSpout PRIVATE SpoutDX_static d3d11 dxgi)
set_target_properties(VisionStereoSpout PROPERTIES SUFFIX ".addon64")
add_library(VisionGameHook MODULE adapters/vision_hook.cpp)
target_include_directories(VisionGameHook PRIVATE third_party/reshade-main/include src)
target_link_libraries(VisionGameHook PRIVATE d3d11 d3d12 dxgi d3dcompiler)
set_target_properties(VisionGameHook PROPERTIES SUFFIX ".addon64" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_compile_options(VisionGameHook PRIVATE /W4 /EHsc /utf-8 /permissive-)
if(MSVC)
  foreach(t vision_core vision_platform VisionRestoration vision_usb_recover vision_tests vision_firmware vision_rp2040 vision_rp2040_tests)
    target_compile_options(${t} PRIVATE /W4 /EHsc /utf-8 /permissive-)
  endforeach()
endif()
