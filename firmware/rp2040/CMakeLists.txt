cmake_minimum_required(VERSION 3.24)
set(PICO_BOARD waveshare_rp2040_zero CACHE STRING "Target board")
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(VisionEmitter VERSION 0.4.0 LANGUAGES C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
pico_sdk_init()
add_executable(VisionEmitter main.cpp usb_descriptors.cpp ../../src/rp2040_protocol.cpp ../../src/rp2040_wave.cpp)
target_include_directories(VisionEmitter PRIVATE . ../../src)
set(fingerprint "pico-sdk-2.2.0/tinyusb-86ad6e56/GCC-14.3.1")
foreach(source main.cpp usb_descriptors.cpp tusb_config.h ../../src/rp2040_protocol.h ../../src/rp2040_protocol.cpp ../../src/rp2040_wave.h ../../src/rp2040_wave.cpp)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${source}" digest)
  string(APPEND fingerprint "${digest}")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
endforeach()
string(SHA256 fingerprint "${fingerprint}")
string(SUBSTRING "${fingerprint}" 0 16 firmware_id)
target_compile_definitions(VisionEmitter PRIVATE VISION_FIRMWARE_ID="VRP1/0.4.0/${firmware_id}")
target_compile_definitions(VisionEmitter PRIVATE PICO_STACK_SIZE=4096 PICO_CORE1_STACK_SIZE=4096)
target_compile_options(VisionEmitter PRIVATE -Wall -Wextra $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions;-fno-rtti>)
target_link_libraries(VisionEmitter pico_stdlib pico_multicore pico_rand pico_unique_id hardware_pio hardware_watchdog tinyusb_device)
pico_enable_stdio_usb(VisionEmitter 0)
pico_enable_stdio_uart(VisionEmitter 0)
add_custom_command(TARGET VisionEmitter POST_BUILD COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:VisionEmitter> ${CMAKE_CURRENT_BINARY_DIR}/VisionEmitter.bin)
